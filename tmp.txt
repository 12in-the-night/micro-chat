import os
import subprocess
import yaml

# 1. Force single GPU (Colab only gives 1 anyway)
os.environ["CUDA_VISIBLE_DEVICES"] = "0"
os.environ["PYTHONUNBUFFERED"] = "1"
os.environ["XLA_PYTHON_CLIENT_PREALLOCATE"] = "false"

# ⚠️ PASTE YOUR HUGGINGFACE TOKEN HERE
os.environ["HF_TOKEN"] = "hfB"

# 2. Uninstall conflicting Colab JAX plugins (Fixes the CUDA 12/13 crash)
print("Cleaning up old JAX installations...")
subprocess.run(["pip", "uninstall", "-y", "jax", "jaxlib", "jax-cuda12-plugin", "jax-cuda12-pjrt", "jax-cuda13-plugin", "jax-cuda13-pjrt"], check=False)

# 3. Install Dependencies
print("Installing JAX and dependencies...")
subprocess.run(["pip", "install", "-q", "-U", "jax[cuda12]", "flax", "optax", "datasets", "tokenizers", "huggingface_hub", "pyyaml", "gguf"], check=True)

# 4. Clone Repository
print("Cloning repository...")
subprocess.run(["rm", "-rf", "micro-chat"], check=False)
subprocess.run(["git", "clone", "https://github.com/12in-the-night/micro-chat.git"], check=True)
os.chdir("micro-chat")

# 5. Patch Config for Colab T4 (16GB VRAM, No BF16 support)
print("Patching config for single GPU...")
with open("config.yaml") as f:
    cfg = yaml.safe_load(f)

cfg["training"]["batch"] = 256
cfg["training"]["eval_batch"] = 32
cfg["training"]["precision"] = "fp16"

with open("config.yaml", "w") as f:
    yaml.safe_dump(cfg, f, sort_keys=False)

# 6. Run Tokenizer and Training
print("Starting Tokenizer...")
subprocess.run(["python", "-u", "main.py", "--action", "tokenizer"], check=True)

print("Starting Training...")
subprocess.run(["python", "-u", "main.py"], check=True)

# 7. Export to GGUF
print("Exporting to GGUF...")
subprocess.run(["python", "-u", "to_gguf.py"], check=True)

# 8. Upload Results to HuggingFace
print("Uploading to HuggingFace...")
from huggingface_hub import HfApi
try:
    api = HfApi(token=os.environ["HF_TOKEN"])
    REPO = "rgdfbd/micro-chat-pi"
    api.create_repo(REPO, repo_type="model", exist_ok=True)

    for name in ("micro-chat-f16.gguf", "tokenizer.json", "config.yaml"):
        if os.path.exists(name):
            print(f"Uploading {name}...")
            api.upload_file(name, name, REPO, repo_type="model")

    print(f"\n✅ SUCCESS! Model ready at: https://huggingface.co/{REPO}")
except Exception as e:
    print(f"\n⚠️ Upload failed (Check your HF_TOKEN): {e}")
